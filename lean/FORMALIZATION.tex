\begin{itemize}
  \item \textbf{Section~\ref{sec:crosswalk} --- Paper-to-Lean Theorem \& Algorithm Crosswalk}: Mapping between the theorems, lemmas, corollaries, and algorithms of \code{paper.tex} and their Lean~4 declarations, explicitly distinguishing structural/accuracy theorems from \code{CostM} complexity theorems.
  \item \textbf{Class A (\S\ref{sec:class-a}) --- Assumptions (Trust Boundary)}: The two external cited-sampler hypotheses (\code{Stage1Sampler}, \code{Stage2Sampler}), their correspondence with \S 3.3 Lemma~3.3 (\code{\textbackslash label\{lem:two-stage-sampler\}}) and Appendix~B (\code{\textbackslash label\{app:sampler\}}), and their transcription notes vs.\ the printed literature theorems.

  \item \textbf{Class C (\S\ref{sec:class-c}) --- Statement Conventions}: Separation of structural and \code{CostM} theorems (including \code{thm\_main\_decomp}, \code{thm\_main\_decomp\_of\_samplers}, \code{thm\_2d\_decomp\_concrete}, and \code{thm\_2d\_decomp\_uncapped\_concrete}), the two 2D decomposition implementations (\code{Decomp2D.lean} vs.\ \code{DyadicBucketingAndGoodCube.lean}), $0 / 0 = 0$, structural recursion fuel counters (\code{rRectFuel}, \code{rRect2DFuel}, \code{rUncappedRect2DFuel}, \code{rCubeFuel}), the $\mathbb{R}$ / \code{Finset} interface (\code{noncomputable} headline \code{*Tracked} functions and uniform \code{EstimateNorm} accuracy via \code{accExp}), the floor-division identity \code{triDetParamD\_eq\_floor\_div}, \code{ValidCubePiece} empty-subset totalization and entrywise submatrix dominance (\code{cappedABDecomposition\_dominance}) vs.\ \S 1.2 Definition~1.1 (\code{\textbackslash label\{def:pieces\}}) \& Theorem~1.1 (\code{\textbackslash label\{thm:main-decomp\}}), and non-formalized discussion/consequence items.
  \item \textbf{Class D (\S\ref{sec:class-d}) --- Implementation Choices Preserving the Proved Guarantees}: Six implementation choices (uniform $\alpha'$ parameter, fuel totalization, \code{EstimateNorm} cost envelope in \code{Sift}, full-list fold in Algorithm~6, single-witness clearing per detector call in the BMM reduction, and $O(|U||V|+|U|^2)$ declared charge in \code{Prune}) that preserve the proved guarantees and asymptotic complexity bounds.

\section{\texorpdfstring{Class A --- Assumptions (Trust Boundary; \code{BMMGrid/CitedResults.lean} is Authoritative)}{Class A - Assumptions (Trust Boundary; BMMGrid/CitedResults.lean is Authoritative)}}
\label{sec:class-a}

\textbf{External Sampler Hypotheses (\code{BMMGrid/CitedResults.lean}, primary-source excerpts in \code{lean/SOURCES.md}; \S 3.3 Lemma~3.3 \code{\textbackslash label\{lem:two-stage-sampler\}} \& Appendix~B \code{\textbackslash label\{app:sampler\}}).}
The headline theorems (\code{thm\_main\_decomp\_of\_samplers}, \code{thm\_main\_triangle\_detection}, \code{thm\_main\_triangle\_detection\_graph}, \code{thm\_main\_bmm}; and \code{thm\_main\_3d\_decomposition} via the \code{TwoStageSamplerOracle} they discharge) take two external literature sampler schemas over sample sequences with repeats (\code{List (List (Fin N))} and \code{List (List (Fin (2 \^{} m)))}, sequence average \code{expectSeq}, and two-sided accuracy predicate \code{IsSeqSampler}):

\begin{enumerate}
  \item \textbf{\code{Stage1Sampler}} (\code{[AFKLM24, Lemma 4.4, arXiv:2311.09095v2, p.~17]} = \code{[GW97, STOC 1994, pp.~574--583; Random Structures \& Algorithms 11(4):315--343, 1997, Section~3]}): $(1/e, 1/\delta)$-oblivious sampler on \code{Fin N} with:
  \begin{itemize}
    \item Uniform sequence length: $\code{len N e }\delta \le c \cdot (e \cdot \delta + 1)^k$.
    \item Family size: $\code{(fam N e }\delta\code{).length} \le N \cdot c \cdot (e \cdot \delta + 1)^k$.
    \item \code{CostM} construction cost (for $1 \le N$): $\code{(famTracked N e }\delta\code{).2} \le N \cdot c \cdot (e \cdot \delta \cdot (\code{Nat.log 2 N} + 1) + 1)^k$. Here the $1 \le N$ guard avoids the $N = 0$ degenerate case where $\code{(famTracked 0 e }\delta\code{).2} \ge 1$ for any combinator-built \code{CostM} definition, and the $(\code{Nat.log 2 N} + 1)$ factor in the polynomial base generously absorbs the $\operatorname{poly}(\log N, e, \delta)$ per-step expander-walk and arithmetic overhead of \code{[GW97]} / \code{[AFKLM24, Lemma 4.4]}.
  \end{itemize}

  \item \textbf{\code{Stage2Sampler}} (\code{[GUV09, Theorem 1.5 (and Theorem 5.12), JACM 56(4), Article 20, pp.~20:1--20:34 (p.~4 / Article p.~20:4), 2009]} + \code{[Vadhan 2012, Corollary 6.24(1), printed p.~186 / PDF p.~189]}; Appendix~B also cites \code{[Zuckerman 1997, Lemma 2.6 \& Proposition 2.7]} for the extractor-to-sampler equivalence and \code{[Zuckerman 1997, Theorem 3.17, p.~15]} for historical explicit-sampler context): $(1/e, 1/\delta)$-averaging sampler on \code{Fin (2 \^{} m)} ($\cong \{0,1\}^m$) with:
  \begin{itemize}
    \item Uniform sequence length (logarithmic in $\delta$): $\code{len m e }\delta \le c \cdot ((m + 1) \cdot e \cdot (\code{Nat.log 2 }\delta + 1))^k$.
    \item Family size: $\code{(fam m e }\delta\code{).length} \le c \cdot (2^m \cdot e \cdot \delta + 1)^k$.
    \item \code{CostM} construction cost: $\code{(famTracked m e }\delta\code{).2} \le c \cdot (2^m \cdot e \cdot \delta + 1)^k$. Here the single existential exponent $k \in \mathbb{N}$ absorbs the $O(1/\alpha)$ constant exponent from \code{[GUV09, Theorem 1.5]} (the extractor-based sampler has $2^{O(m + \log\delta)}$ seeds and $\operatorname{poly}(m, e, \log\delta)$ samples, so its construction cost is $\operatorname{poly}(2^m, e, \delta)$).
  \end{itemize}

  \item \textbf{Primary-Source Verification}: Verbatim primary-source quotations and the 4-line derivation of the \code{Stage2Sampler} bounds from \code{[GUV09, Theorem 1.5]} + \code{[Vadhan 2012, Corollary 6.24(1)]} are recorded in the module header of \code{BMMGrid/CitedResults.lean}; direct PDF fetch URLs and SHA-256 digests are recorded in \code{lean/SOURCES.md}.
\end{enumerate}

\textbf{Transcription Notes (\code{(i)}--\code{(v)}, matching \code{BMMGrid/CitedResults.lean}):}
\begin{itemize}
  \item \emph{(i) Uniform sequence length (\code{Stage1Sampler.len\_eq})}: \code{[AFKLM24, Lemma 4.4]} states subset bounds $|T| \le \operatorname{poly}(\varepsilon^{-1}, \delta^{-1})$ without asserting equal lengths across $T \in \mathcal{T}$. Uniform length holds both for the underlying \code{[GW97]} fixed-length random-walk construction and by Appendix~B's generic reduction (running at accuracy $\varepsilon/4$ and padding each subset to a common power-of-two length $N_1 = 2^p$), and is required for the union-bound composition (\code{composeFamilies\_isSeqSampler}) and the common-denominator identity in \code{ApproxGridNorm}.
  \item \emph{(ii) Power-of-two Stage-2 domain (\code{Fin (2 \^{} m)} $\to$ \code{Fin M}, proved)}: \code{Stage2Sampler} is stated over $\{0,1\}^m \cong \operatorname{Fin}(2^m)$; the modulo-projection reduction to arbitrary \code{Fin M} is machine-checked in Lean (\code{stage2FinM\_sampler} in \code{BMMGrid/ObliviousSampler.lean}).
  \item \emph{(iii) Cost bound transcription (\code{cost\_le})}: The sources' polynomial deterministic RAM construction-time claims are transcribed as polynomial \code{CostM} upper bounds with existential constants $c, k \in \mathbb{N}$ (with $1 \le N$ on \code{Stage1Sampler.cost\_le} so $\code{cost\_le}$ does not force $\code{(famTracked 0 e }\delta\code{).2} = 0$); no concrete constant values are claimed.
  \item \emph{(iv) Conditional trust boundary (no inhabitant)}: No Lean inhabitant of \code{Stage1Sampler} or \code{Stage2Sampler} is provided (the $n = 2$ examples in \code{BMMGrid.lean} are conditional sanity checks parameterized by \code{S1 : Stage1Sampler} and \code{S2 : Stage2Sampler}; trivial exhaustive or product-space Chebyshev families do not satisfy the joint linear-in-$N$ family-size, $N$-free sample-length, and $\operatorname{poly}(\log \delta)$ sample-length bounds, so any inhabitant amounts to formalizing \code{[GW97]}/\code{[GUV09]}, which is out of scope); the headline theorems are conditional on the cited results as transcribed above.
  \item \emph{(v) One-sided vs.\ two-sided sampler error}: \code{[Vadhan 2012, Definition 3.29, printed p.~71 / PDF p.~74]} defines averaging samplers with a one-sided bound; the two-sided \code{IsSeqSampler} bound follows ``up to a factor of $2$ in the failure probability $\delta$'' (Vadhan, p.~71, via $1 - f$), absorbed into $c_2, k_2$.
\end{itemize}

\textbf{Machine-Checked Two-Stage Composition (\code{BMMGrid/ObliviousSampler.lean} $\leftrightarrow$ \S 3.3 Lemma~3.3 \code{\textbackslash label\{lem:two-stage-sampler\}} \& Appendix~B \code{\textbackslash label\{app:sampler\}}):}
\begin{itemize}
  \item The composition of \code{S1 : Stage1Sampler} and \code{S2 : Stage2Sampler} into an $(\varepsilon, \delta)$-oblivious sequence sampler with linear family size in $|Y|$ and logarithmic sequence length in $1/\delta$ (\code{twoStage\_compose}, \code{twoStage\_compose\_general}, \code{twoStageConstructTracked}) is machine-checked in Lean; nothing in the composition is assumed.
  \item Set-based \code{IsObliviousSampler} (\code{Defs.lean}), unconditional Chebyshev tuple-sampler constructions (\code{constructChebyshevTupleSampler}, \code{lem\_approx\_grid\_norm\_chebyshev\_tuple}, \code{lem\_dyadic\_sift\_chebyshev} in \code{ObliviousSampler.lean}, \code{ApproxGridNorm.lean}, and \code{MinDegreeAndDyadicSift.lean}; correctness only, with $|\Omega| = |Y|^k$), and the explicit small-universe oracle inhabitant \code{smallTwoStageSamplerOracle} / \code{twoStageSamplerOracle\_nonempty\_of\_le} (\code{ObliviousSampler.lean}, providing an unconditional inhabitant of \code{TwoStageSamplerOracle (Fin n) (1/160)} for all $n \le 5{,}566{,}362{,}550{,}596$) are documented on the \code{auxiliaryRoots} allowlist outside the headline dependency cone.
\end{itemize}

\textbf{Declared-Charge Monad (\code{CostM $\alpha$ := $\alpha \times \mathbb{N}$}, \code{BMMGrid/CostMonadAndLinter.lean}).}
Algorithms~1--6 (\code{Prune}, \code{EstimateNorm}, \code{Sift}, \code{FindRect} / \code{Decompose2D}, \code{FindCube} / \code{Decompose3D}, \code{TriangleDetect}, plus the \S 6 Lemma~6.1 BMM reduction) are formalized as pure total functions in $\code{CostM }\alpha := \alpha \times \mathbb{N}$, paired with untracked specifications and \code{.1} projection equivalences ($\code{fooTracked\_fst : (fooTracked ...).1 = foo ...}$). Cost charges \code{.2} are \textbf{declared combinator charges} (not step counts of a compiled machine binary) organized in two tiers:

\begin{enumerate}
  \item \textbf{Tier A --- Bit-Level Tracked Arithmetic \& List/Table Kernels}:

\section{\texorpdfstring{Class D --- Implementation Choices Preserving the Proved Guarantees}{Class D - Implementation Choices Preserving the Proved Guarantees}}
\label{sec:class-d}

The Lean definitions of Algorithms~1--6 (\code{minDegree} / \code{minDegreeTracked}, \code{concreteApproxGridNormV} / \code{concreteApproxGridNormVTracked}, \code{dyadicSift} / \code{dyadicSiftTracked}, \code{goodRectStep} / \code{aDecomposition} / \code{cappedGoodRect3D} / \code{cappedADecomp3D}, \code{cappedGoodCube} / \code{cappedABDecomposition} / \code{baseCase2Decomposition}, and \code{computableTriangleDetection} / \code{bmmFromTriDet}) adopt six implementation choices relative to the paper's pseudocode, each preserving the structural guarantees and the headline asymptotic complexity bounds; items~2, 4, and 5 also leave the functional outputs unchanged (\code{*\_fst} lemmas), item~1 is a parameter choice (paper \S 8~(iv)) that is at least as accurate as the paper's, and items~3 and 6 are cost-only conventions cross-listed in Class~B:
\begin{enumerate}
  \item \textbf{Uniform \code{EstimateNorm} Accuracy Parameter (Parameter Choice, Not Control Flow)}: In \code{concreteApproxGridNormV} (\code{MinDegreeAndDyadicSift.lean}), Algorithm~2 (\code{EstimateNorm}) is invoked with uniform accuracy $\alpha' = (\varepsilon/32)2^{-3d}$ in place of Algorithm~3 Line~5's $\varepsilon\delta^3/32$ (a parameter choice rather than a control-flow change); since $\delta > 2^{-d}$ at every non-sparse sifting call, $\alpha' \le \varepsilon\delta^3/32$, preserving Lemma~4.2's certification and subrectangle density guarantees.
  \item \textbf{Structural Recursion Fuel \& Well-Founded Termination}: Each \code{while}-loop is totalized via an explicit \code{Nat.ceil} fuel counter (\code{rRectFuel}, \code{rRect2DFuel}, \code{rUncappedRect2DFuel}, \code{rDecompAFuel}, \code{rDecomp2DFuel}, \code{rUncappedDecomp2DFuel}, \code{rCubeFuel}, \code{rSStarFuel}), integer fuel $n^2$ (\code{processBlockTripleLoop}), or well-founded structural recursion \code{termination\_by U'.card} (\code{minDegreeLoop}), paired with sufficiency lemmas showing that the termination condition is reached before fuel exhaustion.
  \item \textbf{Eager \code{EstimateNorm} Binding at \code{Sift} Entry in \code{dyadicSiftTracked} / \code{dyadicSiftConcreteTracked} (Charge, Not Control Flow)}: In \code{dyadicSiftTracked} (\code{MinDegreeAndDyadicSift.lean:1014--1055}, instantiated with \code{concreteApproxGridNormVTracked S X Y A d} in \code{dyadicSiftConcreteTracked} at \code{MinDegreeAndDyadicSift.lean:2299--2302}), \code{V\_tracked} is bound via \code{CostM.bind} at procedure entry before the outer dyadic loop (whereas Algorithm~3 Line~5 invokes \code{EstimateNorm} only after the outer dyadic loop returns \code{none} and $X_{\ge \delta^2}$ is nonempty).   \item \textbf{Full-List Fold \& Omitted Empty-Input Guard in Algorithm~6 (\code{TriangleDetect})}: \code{triangleDetectionOnPieces} and \code{triangleDetectionOnPiecesTracked} (\code{TriangleAndBMM.lean}) omit Algorithm~6 Line~1's explicit empty-input guard (empty vertex sets already yield empty piece scans returning \code{false}) and fold boolean OR (\code{||}) over all pieces in $\mathcal{S}$ rather than terminating early upon the first piece where \code{detectOnPiece} returns \code{true}; the boolean output and worst-case cost bound are unchanged.
  \item \textbf{Single-Witness Clearing per Detector Call in Lemma~6.1 BMM Reduction}: \code{processBlockTripleLoop} and \code{processBlockTripleLoopTracked} (\code{TriangleAndBMM.lean}) locate and clear one triangle witness $(x, z)$ per \code{true} return of \code{computableTriangleDetection} (with integer fuel $n^2$ per block triple in \code{foldBlockTriples}, executing at most $\operatorname{nnz}(C) \le n^2$ witness clears globally by \code{nnz\_C\_strictly\_decreases\_on\_triangle}), preserving exact BMM output equality (\code{bmmFromTriDet\_correct}) and the $O(n^3 / 2^{\Omega(\sqrt{\log n})})$ total cost bound (\code{lem\_bmm\_to\_tridet\_time\_bound}).
  \item \textbf{Iterative Peeling Charge in Algorithm~1 (\code{Prune} / \code{minDegree}; Charge, Not Control Flow)}: \code{minDegree} / \code{minDegreeTracked} (\code{MinDegreeAndDyadicSift.lean}) matches Algorithm~1's iterative ``while $\exists u$ \dots\ remove a minimum-degree vertex'' loop directly (a Class~B declared-charge choice, not a control-flow deviation), charging $O(|U||V| + |U|^2)$ across at most $|U|$ rounds (\code{minDegreeLoopTracked}, \code{minDegreeTracked\_snd\_le}) versus the paper's $O(|U||V|)$ bound (Lemma~3.2); the pruned subset $(U', \text{tag})$ and the $O(n^2)$ step envelope are unchanged.
\end{enumerate}
